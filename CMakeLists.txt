cmake_minimum_required(VERSION 3.20)
project(ynnpack_model_lab LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(LAB_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/.deps" CACHE PATH "Bootstrapped dependency directory")
foreach(dep XNNPACK slinky cpuinfo)
  if(NOT EXISTS "${LAB_DEPS}/${dep}/CMakeLists.txt")
    message(FATAL_ERROR "Run python3 tools/bootstrap.py before configuring")
  endif()
endforeach()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CPUINFO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_MOCK_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SLINKY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(YNNPACK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YNNPACK_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(YNN_ENABLE_CPUINFO ON CACHE BOOL "" FORCE)
# SME is outside this first experiment. Keep the choice explicit and reversible.
option(LAB_ENABLE_SME "Compile YNNPACK SME/SME2 paths" OFF)
set(YNN_ENABLE_ARM64_SME ${LAB_ENABLE_SME} CACHE BOOL "" FORCE)
set(YNN_ENABLE_ARM64_SME2 ${LAB_ENABLE_SME} CACHE BOOL "" FORCE)
# NDK r28 does not expose FP8 intrinsics. This model has no FP8 tensors.
set(YNN_ENABLE_ARM64_NEONFP8 OFF CACHE BOOL "" FORCE)
set(YNN_ENABLE_ARM64_NEONFP8DOT4 OFF CACHE BOOL "" FORCE)
include_directories("${LAB_DEPS}/XNNPACK" "${LAB_DEPS}/slinky")
add_subdirectory("${LAB_DEPS}/cpuinfo" deps/cpuinfo EXCLUDE_FROM_ALL)
add_subdirectory("${LAB_DEPS}/slinky" deps/slinky EXCLUDE_FROM_ALL)
add_subdirectory("${LAB_DEPS}/XNNPACK/ynnpack" deps/ynnpack EXCLUDE_FROM_ALL)
add_library(lab_recurrent_cpu STATIC runtime/ynnpack_recurrent_cpu.cc)
if(NOT MSVC)
  target_compile_options(lab_recurrent_cpu PRIVATE -O3 -ffp-contract=off)
endif()
add_library(lab_runtime INTERFACE)
target_include_directories(lab_runtime INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/common/include")
target_link_libraries(lab_runtime INTERFACE lab_recurrent_cpu ynnpack_composites ynnpack slinky_thread_pool_impl)
if(NOT MSVC)
  # Profiling derives a statement visitor from Slinky's RTTI-free library.
  target_compile_options(lab_runtime INTERFACE -ffp-contract=off -fno-rtti)
else()
  target_compile_options(lab_runtime INTERFACE /GR-)
endif()
# The benchmark frontend is shared by model runners and the weights-free probe.
option(LAB_ENABLE_TOKENIZERS "Enable native HF/SentencePiece text inputs" ON)
option(LAB_BUILD_DIAGNOSTICS "Build offline compiler/plan tools and their integration tests" OFF)
if(LAB_ENABLE_TOKENIZERS)
  include(cmake/Tokenizers.cmake)
endif()
option(LAB_BUILD_DOT_BENCHMARK "Build the one-row ARM low-bit dot experiment" OFF)
if(LAB_BUILD_DOT_BENCHMARK)
  add_executable(dot_decode_bench tools/dot_decode_bench.cc)
  target_link_libraries(dot_decode_bench PRIVATE lab_runtime)
  target_compile_options(dot_decode_bench PRIVATE -O2)
endif()
option(LAB_BUILD_ATTENTION_BENCHMARK "Build weights-free attention layout experiments" OFF)
set(LAB_GGML_ROOT "" CACHE PATH "llama.cpp source tree for matched recurrent CPU benchmarks")
set(LAB_GGML_LIB_DIR "" CACHE PATH "Existing ggml shared-library directory")
set(LAB_RECURRENT_STEP_GENERATED "" CACHE PATH "Standalone generated recurrent-step experiment artifacts")
set(LAB_RECURRENT_FOLLOWUP_GENERATED "" CACHE PATH "Standalone layout/chunk experiment artifacts")
if(LAB_RECURRENT_STEP_GENERATED)
  add_executable(recurrent_step_bench tools/recurrent_step_bench.cc)
  target_include_directories(recurrent_step_bench PRIVATE "${LAB_RECURRENT_STEP_GENERATED}")
  target_link_libraries(recurrent_step_bench PRIVATE lab_runtime)
  target_compile_options(recurrent_step_bench PRIVATE -O3 -ffp-contract=off)
  add_executable(recurrent_step_test tools/recurrent_step_test.cc)
  target_include_directories(recurrent_step_test PRIVATE "${LAB_RECURRENT_STEP_GENERATED}")
  target_link_libraries(recurrent_step_test PRIVATE lab_runtime)
  target_compile_options(recurrent_step_test PRIVATE -O3 -ffp-contract=off)
  add_executable(recurrent_step_diagnostics tools/recurrent_step_bench.cc tools/recurrent_step_diagnostics.cc)
  target_include_directories(recurrent_step_diagnostics PRIVATE "${LAB_RECURRENT_STEP_GENERATED}")
  target_link_libraries(recurrent_step_diagnostics PRIVATE lab_runtime)
  target_compile_options(recurrent_step_diagnostics PRIVATE -O3 -ffp-contract=off)
  target_compile_definitions(recurrent_step_diagnostics PRIVATE LAB_STEP_DIAGNOSTICS=1)
  if(ANDROID)
    set(step_dot_symbol "_ZN3ynn14get_dot_kernelERKNS_8dot_typeERKNS_15dot_cost_modelsERKNS_9dot_shapeENS_16dot_packed_shapeEjNSt6__ndk18optionalIbEEmi")
  else()
    set(step_dot_symbol "_ZN3ynn14get_dot_kernelERKNS_8dot_typeERKNS_15dot_cost_modelsERKNS_9dot_shapeENS_16dot_packed_shapeEjSt8optionalIbEmi")
  endif()
  target_link_options(recurrent_step_diagnostics PRIVATE "-Wl,--wrap=${step_dot_symbol}")
  if(LAB_RECURRENT_FOLLOWUP_GENERATED)
    foreach(target recurrent_step_bench recurrent_step_diagnostics)
      target_include_directories(${target} PRIVATE "${LAB_RECURRENT_FOLLOWUP_GENERATED}")
      target_compile_definitions(${target} PRIVATE LAB_RECURRENT_FOLLOWUP=1)
    endforeach()
  endif()
  if(LAB_GGML_ROOT AND LAB_GGML_LIB_DIR)
    target_compile_definitions(recurrent_step_bench PRIVATE LAB_STEP_GGML=1)
    target_include_directories(recurrent_step_bench PRIVATE "${LAB_GGML_ROOT}/ggml/include")
    target_link_directories(recurrent_step_bench PRIVATE "${LAB_GGML_LIB_DIR}")
    target_link_libraries(recurrent_step_bench PRIVATE ggml-cpu ggml-base)
    set_target_properties(recurrent_step_bench PROPERTIES BUILD_RPATH "${LAB_GGML_LIB_DIR}")
  endif()
endif()
if(LAB_GGML_ROOT AND LAB_GGML_LIB_DIR)
  foreach(name recurrent_cpu_bench causal_conv_cpu_bench)
    add_executable(${name} tools/${name}.cc)
    target_include_directories(${name} PRIVATE "${LAB_GGML_ROOT}/ggml/include")
    target_link_directories(${name} PRIVATE "${LAB_GGML_LIB_DIR}")
    target_link_libraries(${name} PRIVATE lab_runtime ggml-cpu ggml-base)
    set_target_properties(${name} PROPERTIES BUILD_RPATH "${LAB_GGML_LIB_DIR}")
    if(NOT MSVC)
      target_compile_options(${name} PRIVATE -O3 -ffp-contract=off)
    endif()
  endforeach()
endif()
if(LAB_BUILD_ATTENTION_BENCHMARK)
  add_executable(attention_decode_bench tools/attention_decode_bench.cc)
  target_link_libraries(attention_decode_bench PRIVATE lab_runtime)
  target_compile_options(attention_decode_bench PRIVATE -O2)
endif()
# BF16 arithmetic contracts require the retained rounding patch. The runner
# checks this capability when an HF package is selected at runtime.
file(STRINGS "${LAB_DEPS}/XNNPACK/ynnpack/subgraph/fusion.cc" lab_bf16_fix
  REGEX "bool preserves_intermediate_precision")
if(lab_bf16_fix)
  set(LAB_HAS_BF16_ROUNDING 1)
else()
  set(LAB_HAS_BF16_ROUNDING 0)
endif()
target_compile_definitions(lab_runtime INTERFACE LAB_HAS_BF16_ROUNDING=${LAB_HAS_BF16_ROUNDING})
option(LAB_SANITIZERS "Instrument native adapter and tests with ASan and UBSan" OFF)
if(LAB_SANITIZERS AND NOT MSVC)
  target_compile_options(lab_recurrent_cpu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lab_recurrent_cpu INTERFACE -fsanitize=address,undefined)
  target_compile_options(lab_runtime INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lab_runtime INTERFACE -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(recurrent_cpu_test tests/recurrent_cpu_test.cc)
  target_include_directories(recurrent_cpu_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(recurrent_cpu_test PRIVATE lab_recurrent_cpu)
  add_test(NAME recurrent_cpu COMMAND recurrent_cpu_test)
  if(LAB_BUILD_ATTENTION_BENCHMARK)
    foreach(arithmetic fp32 int8)
      foreach(layout batch rows)
        add_test(NAME attention_${arithmetic}_${layout}_global COMMAND attention_decode_bench
          --arithmetic=${arithmetic} --layout=${layout} --histories=1,17,513,1024,17
          --width=256 --kv-heads=1 --threads=1 --iterations=1 --repetitions=1)
        add_test(NAME attention_${arithmetic}_${layout}_window_multihead COMMAND attention_decode_bench
          --arithmetic=${arithmetic} --layout=${layout} --histories=1,17,511,512,513,1024,17
          --width=512 --kv-heads=2 --threads=4 --window=512 --iterations=1 --repetitions=1)
        if(arithmetic STREQUAL "int8")
          add_test(NAME attention_${arithmetic}_${layout}_probability_rounding COMMAND attention_decode_bench
            --arithmetic=int8 --layout=${layout} --histories=511 --width=512 --kv-heads=2
            --capacity=8448 --bank-seed-stride=8448 --threads=4 --iterations=1 --repetitions=1)
        endif()
      endforeach()
    endforeach()
  endif()
  add_executable(execution_profile_test tests/execution_profile_test.cc)
  target_link_libraries(execution_profile_test PRIVATE lab_runtime)
  add_test(NAME execution_profile COMMAND execution_profile_test)
  add_executable(value_trace_test tests/value_trace_test.cc)
  target_link_libraries(value_trace_test PRIVATE lab_runtime)
  add_test(NAME value_trace COMMAND value_trace_test)
  add_executable(safetensors_assets_test tests/safetensors_assets_test.cc third_party/minijson.cc)
  target_include_directories(safetensors_assets_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME safetensors_assets COMMAND safetensors_assets_test)
  if(LAB_HAS_BF16_ROUNDING)
    add_executable(hf_bf16_test tests/hf_bf16_test.cc third_party/minijson.cc)
    target_link_libraries(hf_bf16_test PRIVATE lab_runtime)
    add_test(NAME hf_bf16 COMMAND hf_bf16_test)
  endif()
  add_executable(model_assets_test tests/model_assets_test.cc third_party/minijson.cc)
  target_include_directories(model_assets_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME model_assets COMMAND model_assets_test)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  add_test(NAME execution_profile_accounting COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_execution_profile.py)
  set_tests_properties(execution_profile_accounting PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME publication COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_publication.py)
  set_tests_properties(publication PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME tokenizer_recipes COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_tokenizer_recipes.py)
  set_tests_properties(tokenizer_recipes PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME benchmark_summary COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_benchmark.py)
  set_tests_properties(benchmark_summary PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME runtime_comparison COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_runtime_comparison.py)
  set_tests_properties(runtime_comparison PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME safetensors COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_safetensors.py)
  set_tests_properties(safetensors PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    ENVIRONMENT "YNNPACK_PLAN_TOOL=${CMAKE_CURRENT_BINARY_DIR}/ynn-plan;YNNPACK_PROGRAM_TOOL=${CMAKE_CURRENT_BINARY_DIR}/ynn-program")
  add_test(NAME asset_recipes COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s tests -p test_assets.py)
  set_tests_properties(asset_recipes PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    ENVIRONMENT "YNNPACK_PLAN_TOOL=${CMAKE_CURRENT_BINARY_DIR}/ynn-plan;YNNPACK_PROGRAM_TOOL=${CMAKE_CURRENT_BINARY_DIR}/ynn-program")
  add_executable(quantized_fc_test tests/quantized_fc_test.cc)
  target_link_libraries(quantized_fc_test PRIVATE lab_runtime)
  add_test(NAME quantized_fc COMMAND quantized_fc_test)
  add_executable(blockwise_fc_test tests/blockwise_fc_test.cc)
  target_link_libraries(blockwise_fc_test PRIVATE lab_runtime)
  add_test(NAME blockwise_fc COMMAND blockwise_fc_test)
endif()
