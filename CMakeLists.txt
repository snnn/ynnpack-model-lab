# Generated functions construct one graph. Compile their bounded source files
# once per artifact set, independently of the runner and backend kernel flags.
set(LAB_BUILDER_OPTIMIZATION "2" CACHE STRING "Optimization level for generated construction sources")
if(NOT LAB_BUILDER_OPTIMIZATION MATCHES "^[0123]$")
  message(FATAL_ERROR "LAB_BUILDER_OPTIMIZATION must be 0, 1, 2, or 3")
endif()
function(lab_link_builders target directory)
  get_filename_component(directory "${directory}" REALPATH)
  # Metadata changes when an artifact set switches from header-only builders to
  # source manifests, including manifests absent at the previous configure.
  if(EXISTS "${directory}/model.json")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${directory}/model.json")
  endif()
  set(sources)
  foreach(phase prefill decode)
    set(manifest "${directory}/gemma4_${phase}_sources.cmake")
    if(EXISTS "${manifest}")
      include("${manifest}")
      list(APPEND sources ${YNNPACK_BUILDER_SOURCES})
    endif()
  endforeach()
  # Retained header-only experiment controls are still supported.
  if(NOT sources)
    return()
  endif()
  get_target_property(definitions ${target} COMPILE_DEFINITIONS)
  set(adapter_definitions)
  foreach(definition IN LISTS definitions)
    if(definition MATCHES "^LAB_YNN_")
      list(APPEND adapter_definitions "${definition}")
    endif()
  endforeach()
  string(SHA256 identity "${directory}|${adapter_definitions}")
  string(SUBSTRING "${identity}" 0 12 identity)
  set(builder "lab_builder_${identity}")
  if(NOT TARGET ${builder})
    add_library(${builder} STATIC ${sources})
    target_link_libraries(${builder} PUBLIC lab_runtime)
    target_compile_definitions(${builder} PRIVATE ${adapter_definitions})
    if(MSVC)
      if(LAB_BUILDER_OPTIMIZATION STREQUAL "0")
        target_compile_options(${builder} PRIVATE /Od)
      elseif(LAB_BUILDER_OPTIMIZATION STREQUAL "1")
        target_compile_options(${builder} PRIVATE /O1)
      else()
        target_compile_options(${builder} PRIVATE /O2)
      endif()
    else()
      target_compile_options(${builder} PRIVATE -O${LAB_BUILDER_OPTIMIZATION})
    endif()
  endif()
  target_link_libraries(${target} PRIVATE ${builder})
endfunction()

if(LAB_BUILD_HF)
  file(STRINGS "${LAB_DEPS}/XNNPACK/ynnpack/subgraph/fusion.cc" bf16_fix
    REGEX "bool preserves_intermediate_precision")
  if(NOT bf16_fix)
    message(FATAL_ERROR "HF profiles need the BF16 rounding fix; bootstrap a separate directory with --hf-bf16 and set LAB_DEPS")
  endif()
  foreach(profile hf_bf16 hf_bf16_int8_raw_kv hf_bf16_int8_published_kv hf_fp32_int8_published_kv hf_static_int8_published_kv)
    foreach(header gemma4_hf_config.h gemma4_prefill.h gemma4_decode.h)
      if(NOT EXISTS "${LAB_HF_GENERATED}/${profile}/${header}")
        message(FATAL_ERROR "Missing HF builder: ${LAB_HF_GENERATED}/${profile}/${header}")
      endif()
    endforeach()
    add_executable(gemma4_e2b_${profile} models/gemma4/runner.cc)
    target_compile_definitions(gemma4_e2b_${profile} PRIVATE LAB_GEMMA4_HF=1)
    if(profile MATCHES "^hf_(fp32|static)_")
      target_compile_definitions(gemma4_e2b_${profile} PRIVATE LAB_HF_FLOAT_ACTIVATIONS=1)
    endif()
    target_include_directories(gemma4_e2b_${profile} PRIVATE "${LAB_HF_GENERATED}/${profile}")
    target_link_libraries(gemma4_e2b_${profile} PRIVATE lab_runtime lab_benchmark)
    target_compile_options(gemma4_e2b_${profile} PRIVATE -O1)
    lab_link_builders(gemma4_e2b_${profile} "${LAB_HF_GENERATED}/${profile}")
  endforeach()
endif()
